filtered episode count hidden

SQL:
WITH filtered AS (
  SELECT episode.uuid,
         episode.archived,
         episode.episodeStatus,
         episode.id
  FROM SJEpisode episode
  LEFT JOIN SJPodcast podcast
    ON episode.podcast_id = podcast.id
  WHERE 1 = 1 AND episode.archived = 0 AND ( (episode.playingStatus = 2 OR episode.playingStatus = 3) AND episode.fileType LIKE 'video%' AND (episode.episodeStatus = 2 OR episode.episodeStatus = 3 OR episode.episodeStatus = 1 OR episode.episodeStatus = 4 OR episode.episodeStatus = 6 ) AND (episode.duration >= 300 AND episode.duration <= 5459) AND episode.keepEpisode = 1 AND  episode.podcastUuid in (?,?) AND episode.publishedDate > <relative-time>)
),
deduped AS (
  SELECT uuid
  FROM (
    SELECT f.uuid,
           ROW_NUMBER() OVER (
             PARTITION BY f.uuid
             ORDER BY
               CASE WHEN f.episodeStatus = 1 THEN 0 ELSE 1 END,
               f.id ASC
           ) AS rn
    FROM filtered f
  ) t
  WHERE rn = 1
)
SELECT COUNT(*) FROM deduped
